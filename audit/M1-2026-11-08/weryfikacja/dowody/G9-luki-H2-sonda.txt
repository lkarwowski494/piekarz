import { planNotifications } from '../core/notification-plan';
import { planBakeSteps } from '../core/bake-steps';

it('SONDA G9-H2: najgorszy przypadek — 2 trwające, 5 zaplanowanych, 10 zakwasów na blacie z karmieniem za 1 h', () => {
  const now = Date.UTC(2026, 10, 8, 18, 0);
  const sesja = (id: string) => ({ id, recipeName: id, steps: planBakeSteps({ startAt: now, isHighRye: false }), stepIdx: 0, dutchOvenPhase: 'lid_on' as const, timerEndsAt: now + 3600_000 });
  const planned = Array.from({ length: 5 }, (_, i) => ({ id: `p${i}`, recipeName: `p${i}`, plannedStartAt: now + (i + 1) * 86400_000, steps: planBakeSteps({ startAt: now + (i + 1) * 86400_000, isHighRye: false }) }));
  const starters = Array.from({ length: 10 }, (_, i) => ({ id: `z${i}`, name: `z${i}`, storageState: 'ROOM_TEMP' as const, lastFedAt: now - 11 * 3600_000 }));
  const budzace = [...['a', 'b'], ...planned.map((p) => p.id)].map((w) => ({ wypiekId: w, at: now + 2 * 3600_000, starterId: 'z0', starterName: 'z0', recipeName: w }));
  const bez = { sessions: [sesja('a'), sesja('b')], planned, starters, budzace } as never;
  const wynik = planNotifications(bez, now);
  const kroki = wynik.filter((p) => p.kind === 'krok' && !p.id.endsWith('.start'));
  const starty = wynik.filter((p) => p.id.endsWith('.start'));
  const karm = wynik.filter((p) => p.kind === 'karmienie');
  const najpozniejszyKrok = Math.max(...kroki.map((p) => p.triggerAt));
  console.log('SONDA H2 łącznie', wynik.length, 'kroki', kroki.length, 'starty', starty.length, 'karmienia/budzenia', karm.length, 'z popytu', 10 + 7,
    '| najpóźniejszy zachowany krok za h', (najpozniejszyKrok - now) / 3600_000, '| karmienia za 1 h odrzucone', 10 - karm.filter((p) => p.triggerAt === now + 3600_000).length);
  expect(wynik.length).toBeLessThanOrEqual(40);
});
