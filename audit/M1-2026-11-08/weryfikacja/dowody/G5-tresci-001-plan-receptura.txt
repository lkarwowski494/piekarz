import { CATALOG_RECIPES } from '../src/recipes/katalog';
import { calculateDoughRecipe } from '../src/core/domain-rules';
import { planBakeSteps, isDutchOvenStep } from '../src/core/bake-steps';
import { DUTCH_OVEN_PHASES } from '../src/core/baking-phases';
import { powodBezZakwasu } from '../src/lib/plan-receptury';

test('sonda 06-tresci-001', () => {
  const steps = planBakeSteps({ startAt: Date.UTC(2026, 10, 8, 8), isHighRye: false });
  const pre = steps.find((s) => s.category === 'preheat_oven');
  const bake = steps.find((s) => s.category === 'bake');
  console.log('PLAN preheat:', pre?.instructions, '| bake:', bake?.instructions, bake?.durationMin, 'dutch:', bake && isDutchOvenStep(bake));
  console.log('FAZY:', DUTCH_OVEN_PHASES.lid_on.temperatureC, DUTCH_OVEN_PHASES.lid_on.durationMin, DUTCH_OVEN_PHASES.lid_off.temperatureC, DUTCH_OVEN_PHASES.lid_off.durationMin);
  const rows: string[] = [];
  let rozne = 0;
  for (const r of CATALOG_RECIPES) {
    if (powodBezZakwasu(r.formula) !== null) continue;
    const rye = calculateDoughRecipe(r.formula as any).isHighRye;
    const zgodne = !rye && r.bakeTempSteamC === 240 && r.bakeTempBrowningC === 210 && r.bakeTimeSteamMinutes === 20 && r.bakeTimeBrowningMinutes === 25 && r.recommendedVessel === 'dutch_oven';
    if (!rye && !zgodne) rozne++;
    rows.push(`${r.id} | zyto=${rye} | ${r.bakeTempSteamC}/${r.bakeTempBrowningC} C | ${r.bakeTimeSteamMinutes}+${r.bakeTimeBrowningMinutes} min | ${r.recommendedVessel} | plan=240/210,20+25,garnek ${rye ? '(plan zytni)' : zgodne ? 'ZGODNE' : 'ROZNE'}`);
  }
  console.log(rows.join('\n') + `\nNIE-ZYTNIE Z ZAKWASEM ROZNE OD PLANU: ${rozne}`);
});
