import { PasekKciuka } from '../../src/components/stolnica/PasekKciuka';

const otworzPlan = async (tree: renderer.ReactTestRenderer) => { await act(async () => { tree.root.findByType(PrzyciskiProwadzenia).props.onPlan(); }); await settle(); };

describe('SONDA G7 04-002: skok z arkusza Plan wypieku a cofnięcie', () => {

  it('po Zrobione jest Cofnij poprzedni; po stuknięciu wiersza w arkuszu nie ma Przywróć ani Cofnij', async () => {
    const start = Date.now() - 60_000;
    const steps = planBakeSteps({ startAt: start, isHighRye: false });
    setActiveBake({ id: 'bake_g7', recipeName: 'Chleb', startedAt: start, currentStageId: steps[0].id, steps });
    let tree: renderer.ReactTestRenderer | null = null;
    await act(async () => { tree = renderer.create(<StolnicaScreen />); });
    await settle();
    await act(async () => { tree!.root.findByType(PasekKciuka).props.onZrobione(); });
    await settle();
    const przedSkokiem = (await loadBakeSession())!;
    const cofnijPrzed = tree!.root.findByType(PasekKciuka).props.cofnijDostepne;
    // (hint liczony przed skokiem)
    // stuknięcie wiersza nr 5 w liście kroków arkusza (ten sam Pressable co w aplikacji)
    await otworzPlan(tree!);
    const lista = tree!.root.findByType(BakeStepsList);
    const wiersze = [...new Set(lista.findAll((n) => !!n.props.accessibilityState && 'selected' in n.props.accessibilityState && typeof n.props.onPress === 'function').map((n) => n.props.onPress))].map((onPress) => ({ props: { onPress } }));
    const hintWiersza = lista.findAll((n) => typeof n.props.onPress === 'function' && !!n.props.accessibilityHint).length;
    await act(async () => { wiersze[4].props.onPress(); });
    await settle();
    const poSkoku = (await loadBakeSession())!;
    const cofnijPo = tree!.root.findByType(PasekKciuka).props.cofnijDostepne;
    const pasekPrzywroc = tree!.root.findAllByType(PasekCofnijKroku).length;
    console.log(JSON.stringify({ wierszy: wiersze.length, stepIdxPrzed: przedSkokiem.stepIdx, cofnijPrzed, stepIdxPo: poSkoku.stepIdx, cofnijPo, pasekPrzywroc, hintWiersza, arkuszOtwartyPoSkoku: tree!.root.findAllByType(BakeStepsList).length }));
    act(() => tree!.unmount());
    expect(cofnijPrzed).toBe(true);
    expect(poSkoku.stepIdx).toBe(4);
    expect(cofnijPo).toBe(false);
    expect(pasekPrzywroc).toBe(0);
  });

  it('stuknięcie wiersza bieżącego kroku też restartuje zegar od pełnego czasu', async () => {
