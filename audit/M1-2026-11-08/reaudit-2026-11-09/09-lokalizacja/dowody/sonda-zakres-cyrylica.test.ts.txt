/** Sonda reaudytu 09 (grupa C): zakres mas z „кг” w trybie imperialnym (ru, uk, bg, sr). */
import * as fs from 'node:fs';
import { setLang, t, type Lang } from '../src/lib/i18n';
import { setUnitSystem, formatTextUnits } from '../src/lib/unit-converter';
const K = JSON.parse(fs.readFileSync(process.env.K011!, 'utf8')) as Record<string, string>;
it('sonda', () => {
  const out: string[] = [];
  out.push(`formatTextUnits('1–2 кг дрожжей','imperial','1–2 kg drożdży') = ${formatTextUnits('1–2 кг дрожжей', 'imperial', '1–2 kg drożdży')}`);
  out.push(`formatTextUnits('1–2 kg drożdży','imperial') = ${formatTextUnits('1–2 kg drożdży', 'imperial')}`);
  for (const l of ['ru', 'uk', 'bg', 'sr'] as Lang[]) {
    setLang(l); setUnitSystem('imperial');
    for (const id of ['n1', 'n2']) {
      const s = t(K[id]);
      const zle = [...s.matchAll(/(?<![\d.,])\d+(?:[.,]\d+)?[–-]\d+(?:[.,]\d+)?\s*(?:lb|oz)/g)].map((m) => m[0]);
      out.push(`${l} ${id}: ${zle.join(' | ') || 'brak'}`);
    }
  }
  setLang('pl'); setUnitSystem('metric');
  fs.writeFileSync(process.env.OUT!, out.join('\n') + '\n');
});
/** Sonda reaudytu 09 (grupa C): zakresy mas w trybie imperialnym, wszystkie klucze i 35 języków. */
import * as fs from 'node:fs';
import { ACTIVE_LANGS, setLang, t } from '../src/lib/i18n';
import { setUnitSystem } from '../src/lib/unit-converter';
const EN = require('../src/lib/i18n.en').EN as Record<string, string>;
it('sonda', () => {
  const out: string[] = [];
  const klucze = Object.keys(EN).filter((k) => !k.includes('{') && /\d\s*[–-]\s*\d+(?:,\d+)?\s*(kg|g)\b/.test(k));
  out.push(`klucze PL z zakresem mas: ${klucze.length}`);
  for (const l of ACTIVE_LANGS) {
    setLang(l); setUnitSystem('imperial');
    let n = 0; const przyklad: string[] = [];
    for (const k of klucze) {
      const z = [...t(k).matchAll(/(?<![\d.,]|oz|lb)(?<!(?:oz|lb)\s?)\b\d+(?:[.,]\d+)?\s*[–-]\s*\d+(?:[.,]\d+)?\s*(?:lb|oz)/g)].map((m) => m[0]);
      if (z.length) { n++; if (przyklad.length < 3) przyklad.push(z[0]); }
    }
    out.push(`${l}: kluczy z zakresem bez przeliczonej dolnej granicy: ${n}${przyklad.length ? ' np. ' + przyklad.join(' | ') : ''}`);
  }
  setLang('pl'); setUnitSystem('metric');
  fs.writeFileSync(process.env.OUT!, out.join('\n') + '\n');
});
