/** Sonda reaudytu 09 (grupa C): próbka regresji w 6 językach o różnych pismach. Tylko odczyt, wynik do pliku. */
import * as fs from 'node:fs';
import { setLang, t, type Lang } from '../src/lib/i18n';
import { setUnitSystem } from '../src/lib/unit-converter';
import { formatTime } from '../src/lib/format';
import { gotoweKiedy, gotoweNazwa, startKiedy, etykietaPlanuPrzedStartem } from '../src/lib/spirala-etykieta';
import { tekstZaPozno } from '../src/lib/plan-za-pozno';
const EN = require('../src/lib/i18n.en').EN as Record<string, string>;
const K011 = JSON.parse(fs.readFileSync(process.env.K011!, 'utf8')) as Record<string, string>;
const K014: string[] = JSON.parse(fs.readFileSync('docs/lokalizacja/przeklad-m2-014-2026-11-09.json', 'utf8')).pl;
const JEZ: Lang[] = ['tr', 'ru', 'el', 'ja', 'ko', 'es-419'];
const TAG: Record<string, string> = { tr: 'tr-TR', ru: 'ru-RU', el: 'el-GR', ja: 'ja-JP', ko: 'ko-KR', 'es-419': 'es-419' };
const TERAZ = new Date(2026, 10, 9, 10, 0), DALEKO = new Date(2026, 10, 12, 13, 30), NOC = new Date(2026, 10, 10, 1, 30);
it('sonda', () => {
  const out: string[] = [];
  for (const l of JEZ) {
    setLang(l);
    out.push(`\n===== ${l}`);
    setUnitSystem('metric');
    out.push('--- 011 klucze krytyczne (metryczne)');
    for (const [id, k] of Object.entries(K011)) out.push(`${id}: ${t(k)}`);
    out.push('--- 013 tryb imperialny: klucze z °F i MAS-002');
    setUnitSystem('imperial');
    for (const k of Object.keys(EN).filter((x) => /°F/.test(x)).slice(0, 4)) out.push(`F: ${t(k).slice(0, 220)}`);
    out.push(`MAS-002: ${t('Na przykład w Tefal PF220838 nie wolno przekroczyć łącznie 1000 g ciasta, 620 g mąki chlebowej i 10 g suchych drożdży instant.')}`);
    for (const id of ['n1', 'n2']) out.push(`${id} imp: ${t(K011[id])}`);
    setUnitSystem('metric');
    out.push('--- 014 (15 kluczy; = EN oznaczone)');
    for (const k of K014) { const v = t(k); out.push(`${v === EN[k] ? '[=EN] ' : ''}${v.slice(0, 160)}`); }
    out.push('--- 046/047 godzina');
    out.push(`formatTime 13:30 ${TAG[l]}: ${formatTime(DALEKO, TAG[l])} | 01:30: ${formatTime(NOC, TAG[l])}`);
    for (const k of Object.keys(EN).filter((x) => x.includes('{godzina}')).slice(0, 40)) {
      const p: Record<string, string> = {}; for (const m of k.matchAll(/\{(\w+)\}/g)) p[m[1]] = m[1] === 'godzina' ? formatTime(NOC, TAG[l]) : `‹${m[1]}›`;
      const s = t(k, p); if (/\.\.|a las|στις/.test(s.replace(/…/g, ''))) out.push(`PODEJRZANE: ${s}`);
    }
    out.push('--- 048 data odległa i jutro (format według języka aplikacji)');
    out.push(gotoweKiedy(DALEKO, TERAZ, true)); out.push(gotoweNazwa('‹nazwa›', DALEKO, TERAZ)); out.push(startKiedy(DALEKO, TERAZ));
    out.push(etykietaPlanuPrzedStartem({ liczbaKrokow: 3, gotowe: DALEKO, teraz: TERAZ }) ?? '');
    out.push(tekstZaPozno({ zaPozno: true, startMs: DALEKO.getTime() - 3_600_000, najwczesniejMs: DALEKO.getTime() } as never, TERAZ.getTime())!.tresc);
    out.push(gotoweKiedy(new Date(2026, 10, 10, 13, 30), TERAZ, true));
  }
  setLang('pl'); setUnitSystem('metric');
  fs.writeFileSync(process.env.OUT!, out.join('\n') + '\n');
});
