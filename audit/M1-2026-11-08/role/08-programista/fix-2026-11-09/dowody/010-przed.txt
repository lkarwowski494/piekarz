    /** Tolerancja sumy udziałów mąk przy wczytaniu receptury: jak import JSON (recipe-sharing.ts, 95–105%). */
    export const TOLERANCJA_SUMY_MAK: readonly [number, number] = [95, 105];

    /**
     * Udziały mąk przeskalowane do sumy 100% (ostatnia mąka domyka sumę); null, gdy suma jest poza 95–105%
     * albo listy nie ma [M1-audytor-generalny-150; 08-programista-502].
     */
    export function normalizujMaki<T extends { percentage: number }>(flours: readonly T[]): T[] | null {
      const suma = flours.reduce((s, f) => s + f.percentage, 0);
      if (flours.length === 0 || suma < TOLERANCJA_SUMY_MAK[0] || suma > TOLERANCJA_SUMY_MAK[1]) return null;
      if (Math.abs(suma - 100) <= 0.01) return flours.map((f) => ({ ...f }));
      const skalowane = flours.map((f) => ({ ...f, percentage: (f.percentage * 100) / suma }));
      const reszta = skalowane.slice(0, -1).reduce((s, f) => s + f.percentage, 0);
      skalowane[skalowane.length - 1].percentage = 100 - reszta;
      return skalowane;
    }

    /**
     * Rachunek receptury z walidacją sumy mąk: błąd walidacji zamiast wyjątku (ekran nie wywraca się).
     * Suma 95–105% liczy się po normalizacji bez błędu; inna suma (> 0) liczy się proporcjonalnie, z błędem
     * `sumaMak` do pokazania [M1-audytor-generalny-150; 08-programista-502].
     */
    export function obliczRecepture(input: DoughRecipeInput): { wynik: DoughRecipeResult; blad: { sumaMak: number } | null; bladMasy: boolean } {
      const suma = input.flours.reduce((s, f) => s + f.percentage, 0);
      const sumaBlad = (n: number) => ({ sumaMak: Math.round(n * 10) / 10 });
      // Masa ciasta ≤ 0 albo nie-liczba: błąd walidacji i wynik zerowy zamiast wyjątku w renderze [audyt-niezalezny D-1]
      if (!(input.targetDoughWeightGrams > 0)) {
        return { wynik: wynikZerowy(input), blad: normalizujMaki(input.flours) ? null : sumaBlad(suma), bladMasy: true };
      }
      const wTolerancji = normalizujMaki(input.flours);
      if (wTolerancji) return { wynik: calculateDoughRecipe({ ...input, flours: wTolerancji }), blad: null, bladMasy: false };
      const proporcjonalnie = suma > 0 ? input.flours.map((f) => ({ ...f, percentage: (f.percentage * 100) / suma })) : [{ name: '—', percentage: 100 }];
      return { wynik: calculateDoughRecipe({ ...input, flours: proporcjonalnie }), blad: sumaBlad(suma), bladMasy: false };
    }

    function wynikZerowy(input: DoughRecipeInput): DoughRecipeResult {
      return {
        targetDoughWeightGrams: 0, baseFlourWeightGrams: 0,
        flours: input.flours.map((f) => ({ name: f.name, percentage: f.percentage, weightGrams: 0 })),
        waterGrams: 0, saltGrams: 0, starterGrams: 0, starterFlourGrams: 0, starterWaterGrams: 0, yeastGrams: 0,
        totalFlourGrams: 0, totalWaterGrams: 0, overallHydrationPercent: 0, sumCheckGrams: 0, isHighRye: false, ryePercentage: 0,
      };
    }

    /**
     * RD-00 & RD-02: True Baker's Percentage & Mass Balance Scaling
     * Scales dough recipe strictly to the desired total dough weight without intermediate rounding drift.
     */
    export function calculateDoughRecipe(input: DoughRecipeInput): DoughRecipeResult {
      if (input.targetDoughWeightGrams <= 0) {
        throw new Error('Docelowa waga ciasta musi być większa od zera');
      }

      // Validate flour blend sum
      const flourSumPercent = input.flours.reduce((sum, f) => sum + f.percentage, 0);
      if (Math.abs(flourSumPercent - 100.0) > 0.01) {
        throw new Error(`Suma procentów mąk musi wynosić 100% (aktualnie: ${flourSumPercent.toFixed(1)}%)`);
      }

      const starterHydr = input.starterHydration ?? 100.0;
      const yeastPct = input.extraYeastPercentage ?? 0.0;

      // Total baker's percentage sum:
      // Base Flour (100%) + Water% + Salt% + Starter% + Yeast%
      const totalFormulaPercentage = 100.0 + input.waterPercentage + input.saltPercentage + input.starterPercentage + yeastPct;

      // Base flour mass in grams
      const baseFlourWeight = input.targetDoughWeightGrams / (totalFormulaPercentage / 100.0);

      // Calculate each component
      const waterWeight = (baseFlourWeight * input.waterPercentage) / 100.0;
      const saltWeight = (baseFlourWeight * input.saltPercentage) / 100.0;
      const starterWeight = (baseFlourWeight * input.starterPercentage) / 100.0;
      const yeastWeight = (baseFlourWeight * yeastPct) / 100.0;

      // Breakdown of starter into flour and water:
      // starterHydration = (starterWater / starterFlour) * 100 => starterWeight = starterFlour * (1 + starterHydr / 100)
      const starterFlour = starterWeight / (1.0 + starterHydr / 100.0);
      const starterWater = starterWeight - starterFlour;

      // Total flour & water in batch
      const totalFlour = baseFlourWeight + starterFlour;
      const totalWater = waterWeight + starterWater;

      // True Overall Hydration
      const overallHydration = (totalWater / totalFlour) * 100.0;

      // Detect rye ratio
      const ryePercentage = input.flours
        .filter((f) => f.name.toLowerCase().includes('żyt') || (f.type && f.type.toLowerCase().includes('zytn')))
        .reduce((sum, f) => sum + f.percentage, 0);
      // Wariant żytni: więcej niż 50% mąki żytniej (ZYT-GRA-001; było >= 50) [M1-audytor-generalny-030]
      const isHighRye = ryePercentage > RYE_VARIANT_THRESHOLD_PCT;

      const saltRounded = Number(saltWeight.toFixed(1));
      const yeastRounded = Number(yeastWeight.toFixed(1));
      // [mąka bazowa, woda, zakwas]: wartość dokładna i zaokrąglona do 1 g
      const exact = [baseFlourWeight, waterWeight, starterWeight];
      const grams = exact.map((g) => Math.round(g));

      // DEC-11 (a), bloker BRD C4a nr 1 [M1-audytor-generalny-040]: suma pokazanych gramów (sól i drożdże
      // z 0,1 g) ma się różnić od masy docelowej o ≤ 1 g. Gdy różnica jest większa, korekta ±1 g składnika
      // o największej reszcie zaokrąglenia w potrzebnym kierunku; każdy składnik zostaje < 1 g od wartości dokładnej.
      const dosesTenths = Math.round(saltRounded * 10) + Math.round(yeastRounded * 10);
      const shownTenths = () => 10 * (grams[0] + grams[1] + grams[2]) + dosesTenths;
      const offTenths = Math.round(input.targetDoughWeightGrams * 10) - shownTenths();
      if (Math.abs(offTenths) > 10) {
        const dir = Math.sign(offTenths);
        const errs = grams.map((g, i) => (g - exact[i]) * dir);
        grams[errs.indexOf(Math.min(...errs))] += dir;
      }
      const [baseFlourRounded, waterRounded, starterRounded] = grams;

      // Mąki osobno (karta „Do miski” pokazuje każdą): metoda największej reszty, więc suma wierszy mąk
      // równa się mące bazowej po korekcie DEC-11, a każda mąka jest < 1 g od wartości dokładnej.
      const flourExact = input.flours.map((f) => (baseFlourWeight * f.percentage) / 100.0);
      const flourGrams = flourExact.map((g) => Math.floor(g));
      const reszta = (i: number) => flourExact[i] - flourGrams[i];
      let flourLeft = baseFlourRounded - flourGrams.reduce((a, g) => a + g, 0);
      const byRemainder = flourExact.map((_, i) => i).sort((i, j) => reszta(j) - reszta(i));
      for (const i of byRemainder) {
        if (flourLeft <= 0) break;
        flourGrams[i] += 1;
        flourLeft -= 1;
      }
      const floursCalculated: FlourItemCalculated[] = input.flours.map((f, i) => ({
        name: f.name,
        percentage: f.percentage,
        weightGrams: flourGrams[i],
      }));

      const sumCheck = Math.round(shownTenths() / 10);

      return {
        targetDoughWeightGrams: input.targetDoughWeightGrams,
        baseFlourWeightGrams: baseFlourRounded,
        flours: floursCalculated,
        waterGrams: waterRounded,
        saltGrams: saltRounded,
        starterGrams: starterRounded,
        starterFlourGrams: Math.round(starterFlour),
        starterWaterGrams: Math.round(starterWater),
        yeastGrams: yeastRounded,
        totalFlourGrams: Math.round(totalFlour),
        totalWaterGrams: Math.round(totalWater),
        overallHydrationPercent: Number(overallHydration.toFixed(2)),
        sumCheckGrams: sumCheck,
        isHighRye,
        ryePercentage,
      };
    }

    /**
     * RD-05: Desired Dough Temperature (DDT) Calculation
     * Calculates required water temperature using 4-factor formula (with starter) or 3-factor (without starter).
     */
    export function calculateDDTWaterTemp(settings: DDTSettings): DDTResult {
      const hasStarter = settings.starterTemp !== undefined;
      let targetWater: number;

      if (hasStarter) {
        // 4-factor formula
        targetWater = 4 * settings.targetDDT - (settings.flourTemp + settings.roomTemp + settings.starterTemp! + settings.frictionFactor);
      } else {
        // 3-factor formula
        targetWater = 3 * settings.targetDDT - (settings.flourTemp + settings.roomTemp + settings.frictionFactor);
      }

      const roundedWater = Number(targetWater.toFixed(1));

      if (roundedWater > 45.0) {
        return {
          waterTempTarget: roundedWater,
          status: 'too_hot_danger',
          warningMessagePl: 'Obliczona temperatura wody przekracza 45°C. Grozi to ugotowaniem drożdży i bakterii kwasu mlekowego. Schłodź mąkę lub obniż temperaturę otoczenia.',
        };
      }

      if (roundedWater < 4.0) {
        return {
          waterTempTarget: roundedWater,
          status: 'requires_ice',
          warningMessagePl: 'Obliczona temperatura wody jest poniżej 4°C. Aby uzyskać tę temperaturę bez zamrażania, zastąp część wody lodem.',
        };
      }

      return {
        waterTempTarget: roundedWater,
        status: 'normal',
      };
    }

    /**
     * RD-07: Ice Calculation for Cold Water Requirement
     * Calculates grams of ice to melt in tap water to reach target low water temperature.
     * Formula: IceGrams = TotalWater * (TapTemp - TargetTemp) / (80 + TapTemp)
     */
    export function calculateIceRequirement(
      totalWaterGrams: number,
      tapWaterTempC: number,
      targetWaterTempC: number
    ): IceCalculationResult {
      if (targetWaterTempC >= tapWaterTempC) {
        return {
          iceGrams: 0,
          waterGrams: totalWaterGrams,
          totalWaterGrams,
        };
      }

      const latentHeatFactor = 80.0 + tapWaterTempC;
      const ice = totalWaterGrams * ((tapWaterTempC - targetWaterTempC) / latentHeatFactor);
      const iceRounded = Math.round(ice);
      const remainingWater = totalWaterGrams - iceRounded;

      return {
        iceGrams: iceRounded,
        waterGrams: remainingWater,
        totalWaterGrams,
      };
    }

    /**
     * RD-08: Temperature-Dependent Fermentation Time Adaptation (Q10 = 3,0 w 20–26 °C, FER-Q10-002)
     * Adapts fermentation duration based on actual dough/ambient temperature.
     */
    export function calculateFermentationAdaptation(
      baseHours: number,
      actualTempC: number,
      baseCalibrationTempC = 24.0,
      q10Factor = Q10_FERMENTATION
    ): FermentationAdaptationResult {
      if (baseHours <= 0 || actualTempC <= 0) {
        throw new Error('Czas bazowy i temperatura muszą być większe od zera');
      }

      // Q10 factor formula: time(T) = baseTime * q10Factor^((baseT - actualT) / 10)
      const exponent = (baseCalibrationTempC - actualTempC) / 10.0;
      const factor = Math.pow(q10Factor, exponent);
      const adaptedUnrounded = baseHours * factor;
      const adaptedHours = Number(adaptedUnrounded.toFixed(2));

      const diffHours = adaptedUnrounded - baseHours;
      const rawDiffMinutes = Math.round(diffHours * 60);
      const diffMinutes = rawDiffMinutes === 0 ? 0 : rawDiffMinutes;

      const totalMinutes = Math.round(adaptedHours * 60);
      const h = Math.floor(totalMinutes / 60);
      const m = totalMinutes % 60;
      const formatted = h > 0 ? `${h}h ${m}m` : `${m}m`;

      const rodzaj: KomunikatAdaptacji['rodzaj'] = Math.abs(diffMinutes) < 5 ? 'optymalna' : diffMinutes < 0 ? 'cieplo' : 'chlodno';

      return {
        baseHours,
        adaptedHours: Number(adaptedHours.toFixed(2)),
        adaptedMinutesFormatted: formatted,
        diffMinutes,
        temperatureCelsius: actualTempC,
        komunikat: { rodzaj, temperaturaC: actualTempC, minuty: Math.abs(diffMinutes) },
      };
    }

    /** Świeże → suche (RD-19, D-172); też RD-37. */
    export const WSP_SWIEZE_NA_SUCHE = 0.35;

    /**
     * RD-19: drożdże świeże na suche i odwrotnie (aktywne i instant) × 0,35, odwrotnie ÷ 0,35 (D-205, D-172, RD-19-W2;
     * wcześniej × 0,4 z S-40, S-39 [M1-audytor-generalny-130]). Producent na opakowaniu może podawać inaczej (D-172).
     */
    export function convertYeast(
      amountGrams: number,
      from: 'fresh' | 'instant'
    ): { freshGrams: number; instantGrams: number } {
      if (amountGrams < 0) {
        throw new Error('Masa drożdży nie może być ujemna');
      }

      if (from === 'fresh') {
        const instant = Number((amountGrams * WSP_SWIEZE_NA_SUCHE).toFixed(2));
        return { freshGrams: amountGrams, instantGrams: instant };
      } else {
        const fresh = Number((amountGrams / WSP_SWIEZE_NA_SUCHE).toFixed(1));
        return { freshGrams: fresh, instantGrams: amountGrams };
      }
    }

    /**
     * RD-16: Calculate recipe scaling from given base flour weight
     */
    export function calculateDoughRecipeFromFlour(
      flourGrams: number,
      input: Omit<DoughRecipeInput, 'targetDoughWeightGrams'>
    ): DoughRecipeResult {
      if (flourGrams <= 0) {
        throw new Error('Waga mąki musi być większa od zera');
      }
      const yeastPct = input.extraYeastPercentage ?? 0.0;
      const totalFormulaPercentage = 100.0 + input.waterPercentage + input.saltPercentage + input.starterPercentage + yeastPct;
      const targetDoughWeightGrams = Math.round(flourGrams * (totalFormulaPercentage / 100.0));
      return calculateDoughRecipe({
        ...input,
        targetDoughWeightGrams,
      });
    }

    /**
     * RD-17: Calculate recipe scaling from target number and weight of loaves
     */
    export function calculateDoughRecipeForLoaves(
      loavesCount: number,
      loafWeightGrams: number,
      input: Omit<DoughRecipeInput, 'targetDoughWeightGrams'>
    ): DoughRecipeResult {
      if (loavesCount <= 0 || loafWeightGrams <= 0) {
        throw new Error('Liczba bochenków i gramatura muszą być większe od zera');
      }
      const targetDoughWeightGrams = Math.round(loavesCount * loafWeightGrams);
      return calculateDoughRecipe({
        ...input,
        targetDoughWeightGrams,
      });
    }
