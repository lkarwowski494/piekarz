/** Sonda roli 12 (scratchpad): wektory strażnika startu z wzorców roli 12 (próg 2) na kodzie src/lib/startup-guard.ts. */
import * as FileSystem from 'expo-file-system/legacy';
const mockFiles = (FileSystem as unknown as { __mockFiles: Map<string, string> }).__mockFiles;
const out = (s: string) => process.stdout.write(s + '\n');
const G = () => require('../startup-guard') as typeof import('../startup-guard');
const PLIK = `${FileSystem.documentDirectory}.piekarz/start-guard.json`;
const plik = () => mockFiles.get(PLIK) ?? '(brak)';

describe('sonda strażnika', () => {
  beforeEach(() => mockFiles.clear());
  it('wektory', async () => {
    const wynik: string[] = [];
    const krok = async (opis: string, f: () => Promise<unknown>) => { const r = await f(); wynik.push(`${opis} → ${JSON.stringify(r)} plik=${plik()}`); };
    await krok('pierwszy start', () => G().beginStartup());
    await krok('  kamień milowy', () => G().markStartupSucceeded());
    await krok('start po udanym', () => G().beginStartup());
    await krok('  (bez kamienia) kolejny start = drugi z rzędu bez kamienia?', () => G().beginStartup());
    await krok('  trzeci start', () => G().beginStartup());
    await krok('  start awaryjny i jego ekran: kolejny start', () => G().beginStartup());
    // „spróbuj normalnie” = startAplikacji bez beginStartup i bez zerowania (app/_layout.tsx:117-121); kamień → 0
    await krok('  „spróbuj normalnie” i kamień', () => G().markStartupSucceeded());
    await krok('start po kamieniu', () => G().beginStartup());
    for (const zly of ['{"x":1}', 'nie-json', 'null', '{"failedStarts":-3}', '{"failedStarts":"2"}']) {
      mockFiles.set(PLIK, zly);
      await krok(`zapis ${zly}`, () => G().beginStartup());
    }
    out(wynik.join('\n'));
    expect(true).toBe(true);
  });
});
