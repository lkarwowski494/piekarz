jest.mock('expo-sqlite', () => require('./helpers/node-sqlite-adapter').createExpoSqliteMock());

import * as fs from 'node:fs';
import * as path from 'node:path';

const mockFiles = (FileSystem as unknown as { __mockFiles: Map<string, string> }).__mockFiles;
const APP_DB = () => dbFilePath('piekarz.db', '/Library/SQLite');
const log = (...a: unknown[]) => process.stdout.write(a.map((x) => (typeof x === 'string' ? x : JSON.stringify(x))).join(' ') + '\n');
const kopieSprzed = () => fs.readdirSync(path.dirname(APP_DB())).filter((f) => /^backup_pre_migration.*\.db$/.test(f));
const liczbyBazy = (plik: string) => {
  const r = openRaw(plik);
  const c = tableCounts(r);
  const v = (r.prepare('PRAGMA user_version').get() as { user_version: number }).user_version;
  const ic = (r.prepare('PRAGMA integrity_check').get() as { integrity_check: string }).integrity_check;
  r.close();
  return { v, ic, bakes: c.bakes, starters: c.starters, recipes: c.recipes, bake_sessions: c.bake_sessions, starter_entries: c.starter_entries };
};

/** Jeden start: kod RootLayout (app/_layout.tsx:64-69, 76-80) i StorageGate (renderuje ekrany tylko w 'ok' / 'newer'). */
async function start(nr: number) {
  jest.resetModules();
  const guard = require('../../lib/startup-guard') as typeof import('../../lib/startup-guard');
  const { rescue } = await guard.beginStartup();
  if (rescue) {
    log(`start ${nr}: TRYB AWARYJNY (bez otwierania bazy)`);
    return { rescue: true as const };
  }
  const app = restartApp();
  await app.schema.initDatabase();
  const stan = app.status.getStorageState();
  const kamien = stan === 'ok' || stan === 'newer';
  if (kamien) await guard.markStartupSucceeded();
  log(`start ${nr}: stan=${stan} plikDoUdostepnienia=${app.status.getRescueFile()?.split('/').pop() ?? null} rodzaj=${app.status.getRescueKind()} kamien=${kamien}`);
  return { rescue: false as const, app };
}
