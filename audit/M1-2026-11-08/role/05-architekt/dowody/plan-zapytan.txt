/** Sonda audytu 05-architekt: EXPLAIN QUERY PLAN zapytań głównych ekranów na danych z szacunku 3 lat (scratchpad). */
jest.mock('expo-sqlite', () => require('./helpers/node-sqlite-adapter').createExpoSqliteMock());
import { useTempDbDir, restartApp, openRaw, dbFilePath, tableCounts } from './helpers/node-sqlite-adapter';
const log = (s: string) => process.stdout.write(`SONDA ${s}\n`);
it('plan zapytań', async () => {
  useTempDbDir();
  const app = restartApp();
  await app.schema.initDatabase();
  const raw = openRaw(dbFilePath('piekarz.db', '/Library/SQLite'));
  // Szacunek 3 lat: 2 wypieki/tydzień = ok. 312 wpisów, 3 zakwasy z dokarmieniem raz dziennie = ok. 3300 wpisów dziennika
  raw.exec('BEGIN');
  for (let i = 0; i < 312; i++) raw.prepare('INSERT INTO bakes (id, recipe_name, started_at, finished_at, notes) VALUES (?, ?, ?, ?, ?)').run(`b${i}`, 'Chleb', 1e12 + i * 3.5 * 864e5, 1e12 + i * 3.5 * 864e5 + 864e5, 'notatka '.repeat(20));
  for (let s = 0; s < 3; s++) {
    raw.prepare('INSERT INTO starters (id, name, flour_type, hydration, birth_date, storage_state) VALUES (?, ?, ?, 100, 1, ?)').run(`s${s}`, `Z${s}`, 'Żytnia', 'ROOM_TEMP');
    for (let d = 0; d < 1100; d++) raw.prepare("INSERT INTO starter_entries (id, starter_id, kind, occurred_at, feed_ratio, signs, created_at, updated_at) VALUES (?, ?, 'feeding', ?, '1:1:1', '[]', ?, ?)").run(`e${s}-${d}`, `s${s}`, 1e12 + d * 864e5, 1, 1);
  }
  for (let i = 0; i < 7; i++) raw.prepare('INSERT INTO bake_sessions (id, status, session_json, created_at, updated_at) VALUES (?, ?, ?, ?, ?)').run(`p${i}`, i < 2 ? 'active' : 'planned', '{}', i, i);
  raw.exec('COMMIT; ANALYZE;');
  log(`liczności: ${JSON.stringify(tableCounts(raw))}`);
  const q: Record<string, string> = {
    dziennik: 'SELECT id, recipe_name, started_at FROM bakes ORDER BY started_at DESC',
    zakwasy: `SELECT s.id, (SELECT e.occurred_at FROM starter_entries e WHERE e.starter_id = s.id AND e.kind = 'feeding' ORDER BY e.occurred_at DESC LIMIT 1) AS lastFedAt FROM starters s`,
    wpisyZakwasu: 'SELECT id FROM starter_entries WHERE starter_id = ? ORDER BY occurred_at DESC, id DESC LIMIT ?',
    maxKarmienia: "SELECT max(occurred_at) AS at FROM starter_entries WHERE starter_id = ? AND kind = 'feeding'",
    sesjeTrwajace: "SELECT id FROM bake_sessions WHERE status = ? ORDER BY created_at, id",
    limitSesji: "SELECT count(*) FROM bake_sessions WHERE status = ? AND id <> ? AND json_valid(session_json)",
    receptury: 'SELECT * FROM recipes ORDER BY created_at ASC',
    ustawienie: 'SELECT value FROM app_settings WHERE key = ?',
    wpisPoId: 'SELECT 1 FROM bakes WHERE id = ?',
  };
  for (const [n, sql] of Object.entries(q)) {
    const plan = raw.prepare(`EXPLAIN QUERY PLAN ${sql}`).all().map((r) => (r as { detail: string }).detail);
    log(`${n}: ${plan.join(' | ')}`);
  }
  raw.close();
  const t0 = performance.now();
  await app.store.getBakeLogs(); await app.store.getStarters();
  log(`czas getBakeLogs+getStarters (laptop, Node, 1 pomiar, NIE budżet): ${(performance.now() - t0).toFixed(1)} ms`);
});
