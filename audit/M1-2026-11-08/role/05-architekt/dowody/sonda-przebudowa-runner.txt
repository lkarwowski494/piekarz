const PRZEBUDOWA = async (db: SQLiteDatabase, offWRun: boolean) => {
  if (offWRun) await db.execAsync('PRAGMA foreign_keys = OFF;');
  // Krok 4: nowa tabela w tym samym kształcie (DDL z sqlite_master), krok 5: kopia, 6: DROP, 7: RENAME
  const ddl = (await db.getFirstAsync<{ sql: string }>("SELECT sql FROM sqlite_master WHERE type = 'table' AND name = 'starters'"))!.sql;
  await db.execAsync(ddl.replace(/CREATE TABLE "?starters"?/, 'CREATE TABLE new_starters') + ';');
  await db.execAsync('INSERT INTO new_starters SELECT * FROM starters; DROP TABLE starters; ALTER TABLE new_starters RENAME TO starters;');
};
