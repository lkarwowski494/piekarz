import React from 'react';
import renderer, { act } from 'react-test-renderer';

const flush = async () => { await act(async () => { for (let i = 0; i < 30; i++) await Promise.resolve(); }); };
const wynik: string[] = [];
function zbierz(nazwa: string, tree: renderer.ReactTestRenderer) {
  const teksty = new Set<string>();
  const tekstWezla = (n: renderer.ReactTestInstance): string => n.children.map((c) => (typeof c === 'string' ? c : tekstWezla(c))).join('');
  for (const n of tree.root.findAll((x) => (x.type as unknown) === 'Text')) {
    const s = tekstWezla(n).trim();
    if (/\p{L}{2,}/u.test(s) && !s.includes('⟦')) teksty.add(`tekst: ${s}`);
  }
  for (const n of tree.root.findAll((x) => typeof x.props?.accessibilityLabel === 'string' || typeof x.props?.accessibilityHint === 'string' || typeof x.props?.placeholder === 'string')) {
    for (const p of ['accessibilityLabel', 'accessibilityHint', 'placeholder']) {
      const s = n.props[p];
      if (typeof s === 'string' && /\p{L}{2,}/u.test(s) && !s.includes('⟦')) teksty.add(`${p}: ${s}`);
    }
  }
  wynik.push(`## ${nazwa}: ${teksty.size}`, ...[...teksty].map((x) => `- ${x.slice(0, 160)}`));
}
async function render(el: React.ReactElement) {
  let tree: renderer.ReactTestRenderer | null = null;
  await act(async () => { tree = renderer.create(<AppAccessProvider>{el}</AppAccessProvider>); });
  await flush();
  return tree!;
}

describe('sonda pseudolokalizacji (de → pseudo)', () => {
  beforeEach(async () => { _resetMemStore(); zUprawnieniem(); await initDatabase(); setLang('de'); });
  afterAll(() => { require('fs').writeFileSync(process.env.SONDA_WYNIK ?? '/dev/stdout', wynik.join('\n') + '\n'); });
  for (const [nazwa, el] of EKRANY) {
    it(nazwa, async () => {
      if (nazwa.startsWith('Stolnica')) await zacznijWypiekBezReceptury(Date.now());
      setLang('de');
      const tree = await render(el());
      zbierz(nazwa, tree);
      await act(async () => tree.unmount());
    });
  }
});
