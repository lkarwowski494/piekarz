// Eksport słowników TS (klucz = tekst PL) do JSON: node eksport.cjs <repo> <wyjscie>
const ts = require(require.resolve('typescript', { paths: [process.argv[2]] }));
const fs = require('fs'), path = require('path');
const [repo, out] = process.argv.slice(2);
fs.mkdirSync(out, { recursive: true });
function load(file) {
  const src = fs.readFileSync(file, 'utf8');
  const js = ts.transpileModule(src, { compilerOptions: { module: ts.ModuleKind.CommonJS, target: ts.ScriptTarget.ES2020 } }).outputText;
  const m = { exports: {} };
  new Function('module', 'exports', 'require', js)(m, m.exports, (p) => { if (p.startsWith('.')) { const f = path.resolve(path.dirname(file), p); for (const c of [f + '.ts', f + '/index.ts']) if (fs.existsSync(c)) return load(c); } return {}; });
  return m.exports;
}
const merge = (mod) => Object.assign({}, ...Object.values(mod).filter((v) => v && typeof v === 'object'));
const en = merge(load(path.join(repo, 'src/lib/i18n.en.ts')));
fs.writeFileSync(path.join(out, 'en.json'), JSON.stringify(en, null, 1));
const pl = {}; for (const k of Object.keys(en)) { if (!k.includes('#')) pl[k] = k.split('|')[0]; }
fs.writeFileSync(path.join(out, 'pl.json'), JSON.stringify(pl, null, 1));
const dir = path.join(repo, 'src/lib/i18n-slowniki');
for (const l of fs.readdirSync(dir)) { const p = path.join(dir, l, 'index.ts'); if (!fs.existsSync(p)) continue; const d = load(p).SLOWNIK; fs.writeFileSync(path.join(out, l + '.json'), JSON.stringify(d, null, 1)); console.log(l, Object.keys(d).length); }
console.log('en', Object.keys(en).length, 'pl', Object.keys(pl).length);
