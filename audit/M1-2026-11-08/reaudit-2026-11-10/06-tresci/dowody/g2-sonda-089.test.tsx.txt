const host = (root: renderer.ReactTestInstance, id: string) =>
  root.findAll((n) => typeof n.type === 'string' && n.props.testID === id);
const settle = async () => {
  await act(async () => {

  });

const otworzPlan = async (id: string) => {
  const r = CATALOG_RECIPES.find((x) => x.id === id)!;
  ustawRecepturePlanu({ recipeId: id, recipeName: id, calculatedRecipe: calculateDoughRecipe(r.formula), recipeFormula: r.formula });
  let tree: renderer.ReactTestRenderer | null = null;
  await act(async () => {
    tree = renderer.create(<HarmonogramScreen />);
  });
  await settle();
  return tree!;
};


import * as fsG2 from 'node:fs';
const teksty = (root: renderer.ReactTestInstance): string[] => {
  const out: string[] = [];
  const idz = (n: renderer.ReactTestInstance | string) => {
    if (typeof n === 'string') { if (n.trim()) out.push(n.trim()); return; }
    for (const c of n.children) idz(c as never);
  };
  idz(root);
  return out;
};
describe('sonda G2 089: co widać na planie zablokowanym', () => {
  let tree: renderer.ReactTestRenderer | null = null;
  beforeEach(async () => { useTempDbDir(); zUprawnieniem(); _resetMemStore(); setActiveBake(null); setLang('pl'); ustawRecepturePlanu(null); await initDatabase(); });
  afterEach(() => { if (tree) act(() => tree?.unmount()); tree = null; ustawRecepturePlanu(null); });
  it('zrzut tekstów', async () => {
    tree = await otworzPlan('lievito_madre_panettone_brioche');
    const ids = ['plan-start-bake-btn', 'plan-kalendarz-btn', 'copy-schedule-btn', 'plan-schedule-btn', 'plan-zaplanuj-btn'];
    const stan = ids.map((id) => `${id}: ${host(tree!.root, id).length} szt.${host(tree!.root, id)[0] ? ' ' + JSON.stringify(host(tree!.root, id)[0].props.accessibilityState ?? null) : ''}`);
    const wszystkieId = [...new Set(tree.root.findAll((n) => typeof n.type === 'string' && typeof n.props.testID === 'string').map((n) => n.props.testID as string))];
    fsG2.writeFileSync(process.env.G2_OUT!, ['# Sonda G2 089 — commit 1c723632 — ekran planu (app/(tabs)/receptury/harmonogram.tsx) dla brioszki ustawionej jak z „Piecz ponownie”', '', '## Przyciski', ...stan, '', '## testID na ekranie', wszystkieId.join(', '), '', '## Teksty na ekranie (kolejność drzewa)', ...teksty(tree.root)].join('\n') + '\n');
  });
});
