describe('sonda', () => {
  afterEach(() => { jest.restoreAllMocks(); });
  it('pomiar', async () => {
    const out: string[] = [];
    const idxy = kroki.map((_, i) => i).filter((i) => ostrzezeniaKroku(kroki, i, kroki[i].category === 'bake' ? 'lid_on' : undefined).length > 0);
    out.push('KROKI ' + kroki.map((k, i) => `${i}:${k.id}/${k.category}/${k.name}/instr=${k.instructions.length}/ostrz=${ostrzezeniaKroku(kroki, i, kroki[i].category === 'bake' ? 'lid_on' : undefined).length}`).join(' | '));
    const jezyki = (process.env.JEZ ? process.env.JEZ.split(',') : [...LANGS]) as any[];
    for (const e of [EKRAN, NISKI]) for (const idx of idxy) for (const j of jezyki) for (const faza of (kroki[idx].category === 'bake' ? ['lid_on', 'lid_off'] : ['lid_on'])) {
      FAZA = faza;
      zUprawnieniem(); useTempDbDir(); _resetMemStore(); setActiveBake(null); setLang(j);
      mockParams = { wypiek: 'bake_chleb' };
      jest.spyOn(Date, 'now').mockImplementation(() => now);
      await initDatabase(); await saveBakeSession(null);
      await saveBakeSession(wypiek('bake_chleb', 'Wiejski chleb codzienny na zakwasie', idx));
      act(() => { Dimensions.set({ window: { width: e.szer, height: e.wys, scale: 3, fontScale: 1 }, screen: { width: e.szer, height: e.wys, scale: 3, fontScale: 1 } }); });
      Object.assign(mockWciecia, { top: e.gora, bottom: e.dol });
      const tree = await render();
      const ramki = ukladEkranu(tree.root, e.szer, e.wys, e.dol);
      const jest_ = tree.root.findAll((n) => n.props?.testID === 'step-warning' && typeof n.type === 'string').length > 0;
      const zrob = ramka(tree.root, ramki, 'next-or-finish-step-btn');

      const sv = tree.root.findAll((n) => n.type === 'RCTScrollView')[0];
      const okno = ramki.get(sv)!;
      const tyt = tree.root.findAll((n) => n.props?.testID === 'step-title' && typeof n.type === 'string')[0];
      const o = jest_ ? ramka(tree.root, ramki, 'step-warning') : null;
      const garnek = tree.root.findAll((n) => typeof n.props?.testID === 'string' && /dutch|phase|lid/.test(n.props.testID) && typeof n.type === 'string').map((n) => n.props.testID);
      out.push(`${e.szer}x${e.wys} krok=${idx}:${kroki[idx].id} faza=${faza} ${j} instr=${t(kroki[idx].instructions).length} kartaY=${Math.round(karta.y)} kartaDol=${Math.round(karta.y + karta.h)} ostrzGora=${o ? Math.round(o.y) : '-'} ostrzDol=${o ? Math.round(o.y + o.h) : '-'} oknoDol=${Math.round(okno.y + okno.h)} zrobioneGora=${Math.floor(zrob.y)} ${o ? (o.y >= okno.y + okno.h ? 'CALE_POD' : o.y + o.h > okno.y + okno.h ? 'CZESCIOWO_POD' : 'OK') : 'BRAK_OSTRZ'} tyt=${tyt ? JSON.stringify(tyt.children.join('')) : ''} ${[...new Set(garnek)].slice(0,6).join(',')}`);
      act(() => tree.unmount());
