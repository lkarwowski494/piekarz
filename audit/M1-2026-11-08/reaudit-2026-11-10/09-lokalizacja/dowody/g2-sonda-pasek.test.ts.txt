import * as fs from 'node:fs';
import { setLang, t, ACTIVE_LANGS } from '../src/lib/i18n';
test('pasek', () => {
  const out = ['# Sonda G2 — commit 1c723632 — teksty paska kciuka (układ zwarty d5a6dc4c: trzy przyciski w jednym rzędzie przy oknie < 700 pt) w 35 językach', '# kolumny: język | „Zrobione” | „Cofnij poprzedni” | „Po co ten krok” | najdłuższy wyraz w dwóch przyciskach drugorzędnych (znaki)', ''];
  const rows: [string, number, string][] = [];
  for (const l of ACTIVE_LANGS) {
    setLang(l);
    const a = t('Zrobione'), b = t('Cofnij poprzedni'), c = t('Po co ten krok');
    const naj = [...b.split(/\s+/), ...c.split(/\s+/)].sort((x, y) => [...y].length - [...x].length)[0]!;
    rows.push([`${l} | ${a} | ${b} | ${c} | ${naj} (${[...naj].length})`, [...naj].length, l]);
  }
  setLang('pl');
  out.push(...rows.map((r) => r[0]), '', 'Najdłuższe wyrazy (malejąco): ' + rows.filter((r) => !['ja', 'ko', 'zh-Hant'].includes(r[2])).sort((x, y) => y[1] - x[1]).slice(0, 12).map((r) => `${r[2]} ${r[1]}`).join(', '));
  fs.writeFileSync(process.env.G2_OUT!, out.join('\n') + '\n');
});
